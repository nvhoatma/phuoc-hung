if (tryNextCardImagePreviewUrl(img)) return; var media = img.closest ? img.closest('.table-data-card-media') : null; if (media) media.classList.add('is-broken'); }, true); function tryNextCardImagePreviewUrl(img) { var rawUrls = String(img.getAttribute('data-card-preview-urls') || ''); } } return null; } function getImagePreviewSource(mode) { if (!state || !state.tableData) return null; if (mode === 'form') return getFormImagePreviewSource(); return getDetailImagePreviewSource(); } function getDetailImagePreviewSource() { if (!state.tableData || !state.tableData.headers) return null; var headers = []; var record = {}; for (var i = 0; i < fields.length; i += 1) { var name = fields[i].name; headers.push(name); var control = el.recordForm && el.recordForm.elements ? el.recordForm.elements.namedItem(name) : null; } function isImageFieldName(field) { return /(^|[\s_\-])(image|img|photo|picture|pic|thumbnail|thumb|avatar|logo|visual|hero|cover|media|asset|artwork)([\s_\-]|$)|image\s*url|photo\s*url|url\s*image|url\s*photo/i.test(String(field || '')); } function isLikelyImageUrl(url) { var clean = String(url || '').split('?')[0].split('#')[0]; return /^https?:\/\//i.test(url) && /\.(png|jpe?g|webp|gif|svg|avif|bmp|tiff?)$/i.test(clean); } function extractImagePreviewUrl(value) { var text = String(value || '').trim(); if (!text) return ''; var match = text.match(/https?:\/\/[^\s"'<>]+/i); return match ? match[0].replace(/[),.;]+$/, '') : ''; } function normalizeImagePreviewUrl(url) { var urls = getImagePreviewUrls(url); return urls[0] || url; } function getImagePreviewUrls(url) { var urls = []; var driveId = extractGoogleDriveFileId(url); if (driveId) { var encoded = encodeURIComponent(driveId); urls.push('https://drive.google.com/thumbnail?id=' + encoded + '&sz=w1000'); urls.push('https://drive.google.com/uc?export=view&id=' + encoded); urls.push('https://drive.usercontent.google.com/download?id=' + encoded + '&export=view'); } urls.push(url); return uniqueImagePreviewUrls(urls); } function uniqueImagePreviewUrls(urls) { var seen = {}; var result = []; for (var i = 0; i < urls.length; i += 1) { var item = String(urls[i] || ''); if (!item || seen[item]) continue; seen[item] = true; result.push(item); } return result; } function extractGoogleDriveFileId(url) { var text = String(url || ''); var fileMatch = text.match(/drive\.google\.com\/file\/d\/([^/]+)/i); if (fileMatch && fileMatch[1]) return fileMatch[1]; control.classList.add('has-inline-image-preview'); if (label && label.nextSibling) { control.insertBefore(card, label.nextSibling); } else { control.insertBefore(card, control.firstChild); } } function bindFormImagePreviewEvents(form) { if (form.getAttribute('data-image-preview-bound') === '1') return; if (!target || !target.name) return; if (!isImageFieldName(target.name) && !extractImagePreviewUrl(target.value)) return; window.clearTimeout(window.__imagePreviewFormTimer); window.__imagePreviewFormTimer = window.setTimeout(function () { enhanceImagePreviewSafe('form'); }, 140); } function clearImagePreviewCards(mode) { var cards = document.querySelectorAll('[data-image-preview-card]'); for (var i = 0; i < cards.length; i += 1) { if (!imagePreviewCardMatchesMode(cards[i], mode)) continue; cards[i].parentNode.removeChild(cards[i]); } var panes = document.querySelectorAll('.has-image-preview'); for (var j = 0; j < panes.length; j += 1) { panes[j].classList.remove('has-image-preview'); } var fields = document.querySelectorAll('.has-inline-image-preview'); for (var k = 0; k < fields.length; k += 1) { if (mode === 'form' && fields[k].classList && fields[k].classList.contains('detail-field')) continue; if (mode === 'detail' && fields[k].classList && fields[k].classList.contains('form-control')) continue; fields[k].classList.remove('has-inline-image-preview'); } } function imagePreviewCardMatchesMode(card, mode) { if (!mode) return true; if (mode === 'form') { return card.classList.contains('form-inline-image-preview') || card.classList.contains('form-image-preview'); } if (mode === 'detail') { return card.classList.contains('detail-inline-image-preview') || card.classList.contains('detail-image-preview'); } return true; } function findDetailPreviewField(fieldName) { var fields = document.querySelectorAll('.detail-field'); for (var i = 0; i < fields.length; i += 1) { if (fields[i].getAttribute('data-detail-field') === fieldName) return fields[i]; var label = fields[i].querySelector('span'); if (label && label.textContent === fieldName) return fields[i]; } return null; } function findFormPreviewControl(fieldName) { var controls = document.querySelectorAll('.form-control'); for (var i = 0; i < controls.length; i += 1) { if (controls[i].getAttribute('data-form-field') === fieldName) return controls[i]; var label = controls[i].querySelector('span'); if (label && label.textContent === fieldName) return controls[i]; } return null; } function createImagePreviewCard(candidate, placement) { var card = document.createElement('aside'); card.className = 'image-preview-card ' + placement + '-image-preview'; card.setAttribute('data-image-preview-card', ''); var link = document.createElement('a'); link.className = 'image-preview-frame'; placeholder.textContent = 'Preview'; var img = document.createElement('img'); img.alt = candidate.field + ' preview'; img.setAttribute('data-preview-index', '0'); img.src = candidate.previewUrl; img.setAttribute('loading', 'lazy'); img.setAttribute('decoding', 'async'); img.setAttribute('referrerpolicy', 'no-referrer'); img.addEventListener('load', function () { card.classList.add('image-preview-loaded'); }); img.addEventListener('error', function () { if (tryNextImagePreviewUrl(img, candidate)) return; var field = document.createElement('span'); field.textContent = candidate.field; var action = document.createElement('strong'); action.textContent = 'Open'; meta.appendChild(field); meta.appendChild(action); card.appendChild(link); card.appendChild(meta); return card; } function tryNextImagePreviewUrl(img, candidate) { var urls = candidate.previewUrls || []; var currentIndex = Number(img.getAttribute('data-preview-index') || '0'); var nextIndex = currentIndex + 1; if (!urls.length || nextIndex >= urls.length) return false; img.setAttribute('data-preview-index', String(nextIndex)); img.src = urls[nextIndex]; return true; }
