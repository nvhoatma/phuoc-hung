/* Config App nested table and relationship builder helpers. */ function renderFormNestedRelationshipBuilder(record) { const nestedHeader = findConfigDataHeaderByHints(['nested_table', 'nested_tables', 'related_table', 'related_tables']); if (!nestedHeader) return ''; const nested = getFormNestedTableNames(record); const options = getNestedTableOptions(nested.join(', '), record) .filter((name) => !nested.some((selected) => normalizeClientConfigKey(selected) === normalizeClientConfigKey(name))); const sourceTable = getConfigTargetTable(record); const sourceColumns = getSchemaColumns(sourceTable); if (!sourceTable || !sourceTable.key || !sourceColumns.length) { return `
${ICONS.chevronRight} Nested tables & relationships ${nested.length}
Choose a Page ID first so the app can detect the parent table.
`; } return `
${ICONS.chevronRight} Nested tables & relationships ${nested.length}
${nested.map((name) => renderFormRelationshipRow(sourceTable, sourceColumns, name)).join('')}
`; } function renderFormRelationshipRow(sourceTable, sourceColumns, nestedName) { const targetTable = findBootstrapTableByName(nestedName); const targetColumns = getSchemaColumns(targetTable); const existing = getCachedRelationshipRecord(sourceTable, targetTable); const sourceValue = getRecordValueByHints(existing, ['source_field', 'parent_field', 'from_field', 'key_field', 'field']) || guessRelationshipField(sourceColumns, targetColumns, true); const targetValue = getRecordValueByHints(existing, ['target_field', 'child_field', 'to_field', 'lookup_field', 'related_field']) || guessRelationshipField(targetColumns, sourceColumns, false); return `
${escapeHtml(sourceTable.name || 'Parent')} to ${escapeHtml(targetTable ? targetTable.name : nestedName)}
`; } function getCachedRelationshipRecord(sourceTable, targetTable) { if (!sourceTable || !targetTable) return {}; const data = state.configDataCache['config:relationship']; const rows = data && data.rows ? data.rows : []; const sourceKey = normalizeClientConfigKey(sourceTable.name); const targetKey = normalizeClientConfigKey(targetTable.name); const row = rows.find((entry) => ( normalizeClientConfigKey(getRecordValueByHints(entry.record, ['source_table', 'parent_table', 'from_table', 'table_name', 'table'])) === sourceKey && normalizeClientConfigKey(getRecordValueByHints(entry.record, ['target_table', 'child_table', 'to_table', 'related_table'])) === targetKey )); return row ? row.record : {}; } function getFormNestedTableNames(record) { return uniqueConfigOptions([ ...parseConfigMultiValue(getRecordValueByHints(record, ['nested_table', 'nested_tables', 'related_table', 'related_tables'])), ...getRelatedNestedTableNamesForFormRecord(record), ]); } function getRelatedNestedTableNamesForFormRecord(record) { const sourceTable = getConfigTargetTable(record); if (!sourceTable) return []; const data = state.configDataCache['config:relationship']; const rows = data && data.rows ? data.rows : []; const sourceCandidates = [sourceTable.name, sourceTable.key, sourceTable.apiTableName, sourceTable.sheetName] .filter(Boolean) .map(normalizeClientConfigKey); return rows .filter((row) => { const source = normalizeClientConfigKey(getRecordValueByHints(row.record, ['source_table', 'parent_table', 'from_table', 'table_name', 'table'])); return source && sourceCandidates.includes(source); }) .map((row) => getRecordValueByHints(row.record, ['target_table', 'child_table', 'to_table', 'related_table'])) .filter(Boolean); } function guessRelationshipField(primaryColumns, otherColumns, preferId) { const otherKeys = new Set((otherColumns || []).map(normalizeClientConfigKey)); const exact = (primaryColumns || []).find((column) => otherKeys.has(normalizeClientConfigKey(column))); if (exact) return exact; const idField = (primaryColumns || []).find((column) => /^id$|_id$/i.test(normalizeClientConfigKey(column))); return idField || (preferId ? (primaryColumns || [])[0] : '') || ''; }
