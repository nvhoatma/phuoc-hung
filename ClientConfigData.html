}) .catch((error) => { showError(error); return null; }) .finally(() => { if (showBusy) setLoading(false); if (!force) state.configBootstrapInFlight = null; }); if (!force) state.configBootstrapInFlight = request; return request; } function scheduleConfigAppPreload() { // Config App data should load only after the user explicitly enters Config mode. } function warmConfigItem(configKey) { return Promise.all([ fetchConfigData(configKey, false), ensureConfigDependencies(configKey, false), ]).then(([data]) => { if (state.appMode === 'config' && state.activeConfigKey === configKey) { state.configData = data; } function scheduleConfigAutoSyncRefresh() { if (state.configAutoSyncDone || state.configAutoSyncInFlight) return; state.configAutoSyncInFlight = true; const run = () => { gasCall('getConfigAppData', { skipAutoSync: false }) .then(async (bootstrap) => { const autoSyncChanged = hasConfigAutoSyncChanges(bootstrap && bootstrap.autoSync); state.configPrefetched = false; } if (state.appMode === 'config') { applyInitialConfigSelection(); if (autoSyncChanged && state.activeConfigKey) { await loadConfigItem(state.activeConfigKey, { force: true }); }); }; if ('requestIdleCallback' in window) window.requestIdleCallback(run, { timeout: 1400 }); else window.setTimeout(run, 900); } function hasConfigAutoSyncChanges(autoSync) { return !!autoSync && (Number(autoSync.addedCount || 0) > 0 || Number(autoSync.deletedCount || 0) > 0); } function applyInitialConfigSelection() { if (!state.configBootstrap) return; if (!state.activeConfigKey) { const defaultKey = state.configBootstrap.items && state.configBootstrap.items.page ? 'page' : ''; const item = defaultKey ? state.configBootstrap.items[defaultKey] : null; state.activeConfigKey = defaultKey || getFirstConfigItemKey(getFirstConfigTypeKey(), getFirstConfigGroupKey(getFirstConfigTypeKey())); state.configActiveType = item && item.typeKey ? item.typeKey : state.configActiveType || getFirstConfigTypeKey(); state.configActiveGroup = item && item.groupKey ? item.groupKey : state.configActiveGroup || getFirstConfigGroupKey(state.configActiveType); } if (!Object.keys(state.configCollapsedTypes || {}).length) { state.configCollapsedTypes = (state.configBootstrap.types || []).reduce((acc, type) => { acc[type.key] = type.key !== state.configActiveType; return acc; }, {}); } if (!Object.keys(state.configCollapsedGroups || {}).length) { state.configCollapsedGroups = (state.configBootstrap.types || []).reduce((acc, type) => { (type.groups || []).forEach((group) => { acc[`${type.key}:${group.key}`] = type.key !== state.configActiveType || group.key !== state.configActiveGroup; }); return acc; }, {}); } } async function loadConfigItem(configKey, options = {}) { if (!configKey) return; const cacheKey = `config:${configKey}`; if (!options.force && state.configDataCache[cacheKey]) { state.activeConfigKey = configKey; } finally { setLoading(false); } } function ensureConfigDependencies(configKey, forceRefresh) { const requests = []; if ((configKey === 'view' || configKey === 'form' || configKey === 'component') && (forceRefresh || !state.configDataCache['config:page'])) { requests.push(fetchConfigData('page', !!forceRefresh)); } if ((configKey === 'form' || configKey === 'component') && (forceRefresh || !state.configDataCache['config:view'])) { requests.push(fetchConfigData('view', !!forceRefresh)); } if (configKey === 'form' && (forceRefresh || !state.configDataCache['config:relationship'])) { requests.push(fetchConfigData('relationship', !!forceRefresh)); } return requests.length ? Promise.all(requests) : Promise.resolve([]); } function fetchConfigData(configKey, forceRefresh = false) { const cacheKey = `config:${configKey}`; if (!forceRefresh && state.configDataCache[cacheKey]) return Promise.resolve(state.configDataCache[cacheKey]); if (!forceRefresh && state.inFlight[cacheKey]) return state.inFlight[cacheKey]; const request = gasCall('getConfigTableData', configKey, !!forceRefresh) .then((data) => { state.configDataCache[cacheKey] = data; state.configDataCacheMeta[cacheKey] = Date.now(); return data; }) .finally(() => { delete state.inFlight[cacheKey]; }); if (!forceRefresh) state.inFlight[cacheKey] = request; return request; } function scheduleConfigPrefetch() { if (state.configPrefetching || state.configPrefetched || !state.configBootstrap) return; state.configPrefetching = true; const run = () => prefetchConfigItems(); if ('requestIdleCallback' in window) window.requestIdleCallback(run, { timeout: 1200 }); else window.setTimeout(run, 250); } async function prefetchConfigItems() { try { const nearbyKeys = getNearbyConfigItemKeys() .filter((key) => key && key !== state.activeConfigKey && !state.configDataCache[`config:${key}`]); await prefetchConfigQueue(nearbyKeys.slice(0, 2), 1); state.configPrefetched = true; } finally { state.configPrefetching = false; } } function getNearbyConfigItemKeys() { const type = (state.configBootstrap && state.configBootstrap.types ? state.configBootstrap.types : []) .find((entry) => entry.key === state.configActiveType); const group = type ? error.message : error); } } }); await Promise.all(workers); } function waitConfigPrefetchTurn() { return new Promise((resolve) => window.setTimeout(resolve, 1100)); } function getAllConfigItemKeys() { return (state.configBootstrap && state.configBootstrap.types ? state.configBootstrap.types : []) .flatMap((type) => type.groups || []) .flatMap((group) => group.items || []) .map((item) => item.key); } function getDefaultConfigRowNumber(data) { const rows = data && data.rows ? data.rows : []; const eligibleRows = rows.filter((row) => { if (state.activeConfigKey === 'form' && isLegacyConfigFormRow(row.record)) return false; if ((state.activeConfigKey === 'view' || state.activeConfigKey === 'form' || state.activeConfigKey === 'component') && state.configPageFilter) { return getConfigRowPageId(row.record) === state.configPageFilter; } return true; }); const current = eligibleRows.find((row) => row.rowNumber === state.configSelectedRowNumber); return current ? current.rowNumber : eligibleRows[0] ? eligibleRows[0].rowNumber : null; } function filteredConfigRows() { const rows = state.configData && state.configData.rows ? state.configData.rows : []; const query = String(state.configQuery || '').trim().toLowerCase(); return rows.filter((row) => { if (state.activeConfigKey === 'form' && isLegacyConfigFormRow(row.record)) return false; if ((state.activeConfigKey === 'view' || state.activeConfigKey === 'form' || state.activeConfigKey === 'component') && state.configPageFilter) { const pageId = getConfigRowPageId(row.record); if (pageId !== state.configPageFilter) return false; } return !query || row.values.join(' ').toLowerCase().includes(query); }); } function isLegacyConfigFormRow(record) { const pageId = getRecordValueByHints(record, ['page_id', 'page']); if (pageId) return false; const formId = getRecordValueByHints(record, ['form_id', 'id', 'key', 'config_key']); return /\.(table|detail|card)\.form$/i.test(String(formId || '')); } function getConfigPageFilterOptions() { const pageRows = state.configDataCache['config:page'] && state.configDataCache['config:page'].rows ? state.configDataCache['config:page'].rows : []; const cachedViewRows = state.configDataCache['config:view'] && state.configDataCache['config:view'].rows ? state.configDataCache['config:view'].rows : []; const viewRows = state.activeConfigKey === 'view' && state.configData && state.configData.rows ? state.configData.rows : cachedViewRows; const byId = {}; pageRows.forEach((row) => { const id = getRecordValueByHints(row.record, ['page_id', 'page']); if (!id) return; const pageType = normalizeClientConfigKey(getRecordValueByHints(row.record, ['page_type', 'type'])); if (['report', 'external_link', 'embed_link', 'embeded_link'].includes(pageType)) return; byId[id] = { id, label: getRecordValueByHints(row.record, ['item', 'label', 'name', 'title']) || id }; }); viewRows.forEach((row) => { const id = getRecordValueByHints(row.record, ['page_id', 'page']); if (id && !byId[id]) byId[id] = { id, label: id }; }); return Object.keys(byId).sort().map((id) => byId[id]); } function getConfigRowPageId(record) { const pageId = getRecordValueByHints(record, ['page_id', 'page']); if (pageId) return pageId; const viewId = getRecordValueByHints(record, ['view_id', 'view']); const view = getCachedViewRecord(viewId); return view ? getRecordValueByHints(view, ['page_id', 'page']) : ''; } function getCachedViewRecord(viewId) { if (!viewId) return null; const data = state.activeConfigKey === 'view' ? state.configData : state.configDataCache['config:view']; const rows = data && data.rows ? row.record : null; } function getSelectedConfigRow() { const rows = state.configData && state.configData.rows ? state.configData.rows : []; return rows.find((row) => row.rowNumber === state.configSelectedRowNumber) || null; } function getConfigRowTitle(row) { if (state.activeConfigKey === 'dataColumn') { return getRecordValueByHints(row.record, ['column_name', 'field', 'column', 'field_name', 'label', 'item', 'name']) || parseConfigColumnId(getRecordValueByHints(row.record, ['column_id', 'id', 'key'])).column || row.values.find(Boolean) || `Row ${row.rowNumber}`; } if (state.activeConfigKey === 'relationship') { const fromColumn = parseConfigColumnId(getRecordValueByHints(row.record, ['from_column_id', 'from_column', 'source_column_id'])); const toColumn = parseConfigColumnId(getRecordValueByHints(row.record, ['to_column_id', 'to_column', 'target_column_id'])); const sourceField = getRecordValueByHints(row.record, ['source_field', 'parent_field', 'from_field', 'key_field', 'field']) || fromColumn.column; const targetTable = getRecordValueByHints(row.record, ['target_table', 'child_table', 'to_table', 'related_table']) || toColumn.table; const targetField = getRecordValueByHints(row.record, ['target_field', 'child_field', 'to_field', 'lookup_field', 'related_field']) || toColumn.column; if (sourceField || targetField) return `${sourceField || 'Parent'} -> ${targetTable ?
